function equal (a,b: complex): boolean;
begin
    equal := (a.re=b.re) and (a.im=b.im)
end;

function add (a,b: complex): complex;
begin
    add.re := a.re+b.re;
    add.im := a.im+b.im
end;


{ ========================================================================== }
{ Page 115                                                                  }
{ ========================================================================== }

function sub (a,b: complex): complex;
begin
    sub.re := a.re-b.re;
    sub.im := a.im-b.im
end;

function mul (a,b: complex): complex;
begin
    mul.re := (a.re*b.re-a.im*b.im);
    mul.im := (a.re*b.im+a.im*b.re);
end;

function dvd (a,b: complex): complex;
var denom: real;
begin
    denom := sqr(b.re)+sqr(b.im);
    if denom = 0.0 then halt(-5);  (* divide by zero *)
    dvd.re := (b.re*a.re + b.im*a.im) / denom; { RECONSTRUCTED: standard complex division }
    dvd.im := (b.re*a.im - b.im*a.re) / denom;
end;

function conj (a: complex): complex;
begin
    conj.re := a.re;
    conj.im := -a.im
end;

function mag (a:complex): real;
begin
    mag := sqrt(sqr(a.re)+sqr(a.im))
end;

function scmul (scale:real; a:complex): complex;
begin
    scmul.re := scale*a.re;
    scmul.im := scale*a.im
end;

function polar_to_rect(mag,phase:real):complex;
begin
    polar_to_rect.re := mag*cos(phase);
    polar_to_rect.im := mag*sin(phase);
end;

end.  (* complexmath *)

procedure traub_iteration(function g(x,z:real):real; error_limit:real;
                          var input,for_x:real);
var help,E,nå_x,for_y,nå_y,deriv,zett,fzett : real;
begin
    nå_y := g(for_x,input);
    nå_x := for_x*(1.1);
    repeat
        for_y := nå_y;
        nå_y := g(nå_x,input);
        deriv := (nå_y - for_y)/(nå_x - for_x);
        zett := nå_x - (nå_y/deriv);
        fzett := g(zett,input);
        for_x := nå_x;
        help := 2*fzett-nå_y;
        if help=0 then nå_x := zett else
            nå_x := for_x - (for_x-zett)*((fzett-nå_y)/(help));
        if nå_x<>0 then
            E := 1 - (for_x/nå_x)
        else
            if for_x=0 then E:=0 else E:=1;
    until abs(E) < error_limit;


{ ========================================================================== }
{ Page 121                                                                  }
{ ========================================================================== }

    for_x := nå_x;
end;

function f(x:real):real;          (* Global variabel Iq *)
(* Finner base-strømmen utfra utgangsstrømmen *)
var I1,I2,ib1,ib2,limit : real;
begin
    limit := 2*Iq;
    if x >= limit then begin I1:=x; I2:=0; end else
    if x <= -limit then begin I2:=abs(x); I1:=0; end else
                         begin I1:=x/2 + Iq; I2:=Iq - x/2; end;
    ib1 := I1/hfe(I1);
    ib2 := I2/hfe(I2)*offset;
    f := ib1 - ib2;
end;

(*                                                               *)
var ibias : real;
begin
    ibias := antpar*f(x/Rl/antpar);
    g := (ibias - z)*Rg + x;
end;
